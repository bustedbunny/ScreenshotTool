cmake_minimum_required(VERSION 3.24)
project(ScreenshotTool VERSION 1.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ScreenshotTool requires Windows x64 and the Visual Studio C++ toolchain.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
configure_file(src/version.hpp.in generated/version.hpp @ONLY)
configure_file(resources/app.rc.in generated/app.rc @ONLY)
configure_file(resources/app.manifest.in generated/app.manifest @ONLY)
add_library(screenshot_core STATIC src/core.cpp)
target_include_directories(screenshot_core PUBLIC src)
target_compile_definitions(screenshot_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_library(screenshot_services STATIC src/capture.cpp src/graphics.cpp src/export.cpp src/settings.cpp src/inline_text.cpp src/text_store.cpp src/update.cpp)
target_include_directories(screenshot_services PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(screenshot_services PUBLIC screenshot_core d3d11 dxgi dxguid d2d1 dwrite windowscodecs ole32 oleaut32 shell32 shlwapi shcore user32 gdi32 advapi32 comctl32 comdlg32 dwmapi uuid)
target_link_libraries(screenshot_services PUBLIC winhttp bcrypt windowsapp version)
add_executable(ScreenshotTool WIN32 src/main.cpp src/app.cpp src/overlay.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/app.rc")
target_link_libraries(ScreenshotTool PRIVATE screenshot_services)
target_link_options(ScreenshotTool PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/generated/app.manifest")
foreach(target screenshot_core screenshot_services ScreenshotTool)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(ScreenshotToolTests tests/tests.cpp src/overlay.cpp)
  target_link_libraries(ScreenshotToolTests PRIVATE screenshot_services)
  target_compile_options(ScreenshotToolTests PRIVATE /W4 /permissive- /EHsc /utf-8)
  add_test(NAME ScreenshotToolTests COMMAND ScreenshotToolTests)
  add_executable(ScreenshotToolUpdateTests tests/update_tests.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/app.rc")
  target_link_libraries(ScreenshotToolUpdateTests PRIVATE screenshot_services)
  target_compile_options(ScreenshotToolUpdateTests PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_options(ScreenshotToolUpdateTests PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/generated/app.manifest")
  add_test(NAME ScreenshotToolUpdateTests COMMAND ScreenshotToolUpdateTests)
  set_tests_properties(ScreenshotToolUpdateTests PROPERTIES TIMEOUT 90)
endif()
install(TARGETS ScreenshotTool RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
install(FILES docs/VALIDATION.md docs/ARCHITECTURE.md DESTINATION docs)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "ScreenshotTool-${PROJECT_VERSION}-windows-x64")
include(CPack)
